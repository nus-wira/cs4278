\subsection{Circle}
We can sample uniformly $\theta\in[0,2\pi)$. Then

is uniformly sampled from $S$.

\subsection{Disc}
For $0\leq r_0 \leq r$, we want 
\[\Pr[||p-c||\leq r_0] = \frac{r_0^2}{r^2}\]
to sample uniformly in the circle area.\\
We can sample uniformly $\theta\in[0,2\pi),r_0^2\in[0,r^2]$. Then,
\[p=(r_0\cos\theta,r_0\sin\theta)\]
is uniformly sampled from $S$.

\subsection{N-Dimensional Disc}
Rejection sampling can be used where $\forall i \in \{1,2,\dots,N\}$,
sample $x_i\in[0,r]$, where $p=(x_1,x_2,\dots,x_n)$.
Then for sphere with center $c$, reject any $p$ where $||p-c||>r$.
Given the volume of an $N$-dimensional sphere with radius $R$ is \cite{gipple14}
\[V_N(R)=\frac{\pi^{n/2}}{\Gamma\left(\frac{N}{2}+1\right)}R^N\]
where $\Gamma$ is Leonhard Euler's gamma function
\[\Gamma(n+\frac{1}{2})
=(n-\frac{1}{2})\cdot(n-\frac{3}{2})\cdot\dots\cdot\frac{1}{2}\cdot\pi^{1/2}\]
We have the rejection rate
\[\frac{r^N-V_N(r)}{r^N}=1-\frac{\pi^{N/2}}{\Gamma\left(\frac{N}{2}+1\right)}\]
With the recursive formula
\[V_N(R)=\frac{2\pi R^2}{N}V_{N-2}(R)\]
where $R$ is independent of $N$, we can see that
\[\lim_{N\rightarrow\infty}V_N(R)=0\]
leading to higher rejection rates at higher values of $N$.

\section{Rejection Sampling}
First, we can obtain a bounding box for $B$ where,
given $c_i=(c_{ix},c_{iy})$,
\[(x_1,x_2)=(\min_i (c_{ix}-r_i),\max_i (c_{ix}+r_i))\]
\[(y_1,y_2)=(\min_i (c_{iy}-r_i),\max_i (c_{iy}+r_i))\]
and the bounding box $C$ is defined as 
\[C=\{(x,y)\in\mathbb{R}^2 | x_1\leq x \leq x_2 \wedge y_1\leq y \leq y_2\}\]
Sample uniformly $x\in[x_1,x_2],y\in[y_1,y_2]$, for $p=(x,y)$,
then reject any $p$ where $||p-c_i||>r_i$ for all $i=1,2,3$.\\
Depending on how much the 3 circles intersect, the rejection ratio varies.
In the worst case, there is no intersection, so the rejection rate is
\[1-\frac{\pi(r_1^2+r_2^2+r_3^2)}{(x_2-x_1)\cdot(y_2-y_1)}\]
On the other hand, if one circle contains both other circles, the rate is
\[1-\frac{\pi\cdot\max_i r_i^2}{(x_2-x_1)\cdot(y_2-y_1)}\]
In all other cases, there is some intersection between the circles,
so overall, we know the rejection rate $t$ to be
\[\max\{0,1-\frac{\pi(r_1^2+r_2^2+r_3^2)}{(x_2-x_1)\cdot(y_2-y_1)}\}\leq t 
\leq 1-\frac{\pi\cdot\max_i r_i^2}{(x_2-x_1)\cdot(y_2-y_1)}\]

\section{*Heuristic}

\subsection{Admissible}
\begin{equation}
\begin{aligned}
    h(x)&=\max\{h_1(x),h_2(x)\}\\
    &\leq \max\{h^*(x),h^*(x)\} \quad\text{(by definition of admissibility)}\\
    &=h^*(x)
\end{aligned}
\end{equation}
where $h^*(x)$ is the true cost to the goal. $h(x)$ must be admissible.

\subsection{}
\[h(x)\]
Define $f,f_1,f_2$ to be the $f$-costs using $h,h_1,h_2$ 
as the heuristics respectively.
For $h(n)=\max\{h_1(n),h_2(n)\}$, 
we have that $h(n)\geq h_1(n)$ and $h\geq h_2(n)$.
Then, 
\[f(n)=g(n)+h(n)\geq f_1(n)=g(n)+h_1(n)\]
\[f(n)\geq f_2(n)\]
Then any node expanded with $h$ as the heuristic will be expanded by
$h_1$ or $h_2$ as the heuristic.
Suppose the goal node $t$ has $f$-cost $f(t)$. 
Note that $f(t)=f_1(t)=f_2(t)$.
Then, all nodes with $f(n)<f(t)$ will be expanded by A* search.
Thus, using $h$ as the heuristic will minimise the number of nodes
expanded. $h$ also dominates $h_1,h_2$ \cite{russell18}.
